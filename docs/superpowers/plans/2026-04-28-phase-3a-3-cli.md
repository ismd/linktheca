> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `linktheca` CLI binary that talks to the backend over HTTP only, with auth (`register`/`login`/`logout`/`whoami`) and Radar bootstrap (`topic add`/`feed add`/`subscribe`) subcommands. Sessions live in `~/.config/linktheca/session.json` (mode 0600), access tokens auto-refresh on 401, and output toggles between human-readable table and JSON.

**Architecture:** `cmd/linktheca/main.go` registers a cobra root command. CLI-only Go packages live under `cmd/linktheca/internal/cli/` so the backend never transitively pulls cobra. `session/` reads/writes the JSON file. `apiclient/` wraps `*http.Client` with bearer-injection and a `Do` that retries once on 401 after refreshing the access token. `output/` formats Go structs into either `text/tabwriter` tables or pretty JSON. `cmd/` holds one file per subcommand. An end-to-end test under `cmd/linktheca/integration_test.go` builds the binary, starts a real `linktheca-server` in a goroutine with a `FakeEmbedder`, and walks through the full sequence via `os/exec`.

**Tech Stack:** Go 1.26+, `spf13/cobra`, stdlib (`net/http`, `text/tabwriter`, `encoding/json`, `os/exec`), `stretchr/testify`. Reuses the existing test infrastructure: `internal/testing/testdb`, `internal/core/embeddings.FakeEmbedder`. ## Conventions for every task

- **TDD where it adds value.** All non-trivial logic (session, output, auto-refresh) gets unit tests. Cobra subcommands themselves are tiny glue — assert their behavior in the end-to-end test, not in unit tests for `RunE` indirection.
- **Commit after each task.** Stage explicit files; never `git add .`.
- **Run from repo root** (`/home/ismd/coding/linktheca`).
- **Commit messages:** `<type>(<scope>): <subject>` where scope is `cli`, e.g. `feat(cli): add session storage`.
- **Errors:** print `Error: <message>` to stderr; exit 0 on success, 1 on any error. Cobra's default `os.Exit(1)` on `RunE` returning err is fine.
- **No global state.** Subcommand factories take a `*cli.Env` carrying server URL, output format, session loader. The root command builds it from flags + env once.
- **Output:** every command prints either a human table or JSON depending on `--output`. Unknown formats default to table.

### Quick reference

- HTTP endpoints (from phase 3a-2): `POST /auth/register`, `POST /auth/login`, `POST /auth/refresh`, `POST /auth/logout`, `GET /auth/me`, `POST /radar/topics`, `POST /radar/feeds`, `POST /radar/subscriptions`.

// Package cli holds the cobra root and shared environment for the linktheca CLI.
package cli

import (
	"errors"
	"os"

	"github.com/spf13/cobra"
)

// Env carries shared configuration to subcommands.
type Env struct {
	ServerURL    string // resolved server base URL
	ConfigDir    string // resolved config directory; session.json lives inside
	OutputFormat string // "table" | "json"
}

const (
	OutputTable = "table"
	OutputJSON  = "json"
)

// NewRoot builds the cobra root. Subcommands are added by the caller (main.go).
func NewRoot(env *Env) *cobra.Command {
	root := &cobra.Command{
		Use:   "linktheca",
		Short: "Linktheca CLI",
		Long:  "Command-line client for a Linktheca server (auth, radar bootstrap).",
		PersistentPreRunE: func(_ *cobra.Command, _ []string) error {
			if env.ServerURL == "" {
				return errors.New("server URL is empty (set --server or LINKTHECA_URL)")
			}
			if env.OutputFormat != OutputTable && env.OutputFormat != OutputJSON {
				env.OutputFormat = OutputTable
			}
			return nil
		},
		SilenceUsage:  true,
		SilenceErrors: false,
	}
	root.PersistentFlags().StringVar(&env.ServerURL, "server", env.ServerURL,
		"Linktheca server URL (overrides LINKTHECA_URL)")
	root.PersistentFlags().StringVar(&env.ConfigDir, "config", env.ConfigDir,
		"path to config directory (default: $XDG_CONFIG_HOME/linktheca or ~/.config/linktheca)")
	root.PersistentFlags().StringVar(&env.OutputFormat, "output", env.OutputFormat,
		"output format: table | json")
	return root
}

// DefaultEnv builds an Env from environment variables. Flags later override
// these on the cobra command.
func DefaultEnv() *Env {
	url := os.Getenv("LINKTHECA_URL")
	if url == "" {
		url = "http://localhost:8080"
	}
	cfg := os.Getenv("LINKTHECA_CONFIG_DIR")
	if cfg == "" {
		base := os.Getenv("XDG_CONFIG_HOME")
		if base == "" {
			home, _ := os.UserHomeDir()
			base = home + "/.config"
		}
		cfg = base + "/linktheca"
	}
	return &Env{ServerURL: url, ConfigDir: cfg, OutputFormat: OutputTable}
}
```

// Package session reads and writes the CLI's persistent login state
// (~/.config/linktheca/session.json by default, mode 0600).
package session

import (
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"time"
)

const fileName = "session.json"

var ErrNoSession = errors.New("no session")

type Session struct {
	ServerURL    string    `json:"server_url"`
	AccessToken  string    `json:"access_token"`
	RefreshToken string    `json:"refresh_token"`
	UserID       int64     `json:"user_id"`
	IsAdmin      bool      `json:"is_admin"`
	ExpiresAt    time.Time `json:"expires_at"`
}

func Load(dir string) (*Session, error) {
	path := filepath.Join(dir, fileName)
	b, err := os.ReadFile(path)
	if err != nil {
		if errors.Is(err, os.ErrNotExist) {
			return nil, ErrNoSession
		}
		return nil, fmt.Errorf("read session: %w", err)
	}
	var s Session
	if err := json.Unmarshal(b, &s); err != nil {
		return nil, fmt.Errorf("parse session: %w", err)
	}
	return &s, nil
}

func Save(dir string, s *Session) error {
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return fmt.Errorf("mkdir config: %w", err)
	}
	b, err := json.MarshalIndent(s, "", "  ")
	if err != nil {
		return fmt.Errorf("marshal session: %w", err)
	}
	path := filepath.Join(dir, fileName)
	if err := os.WriteFile(path, b, 0o600); err != nil {
		return fmt.Errorf("write session: %w", err)
	}
	// Ensure perms even if file pre-existed with a wider mode.
	if err := os.Chmod(path, 0o600); err != nil {
		return fmt.Errorf("chmod session: %w", err)
	}
	return nil
}

func Clear(dir string) error {
	path := filepath.Join(dir, fileName)
	if err := os.Remove(path); err != nil && !errors.Is(err, os.ErrNotExist) {
		return fmt.Errorf("remove session: %w", err)
	}
	return nil
}
```

```go
// Package output renders Go structs as JSON or as human-readable tables.
// Each field that should appear in the table carries a `header:"..."` tag.
package output

import (
	"encoding/json"
	"fmt"
	"io"
	"reflect"
	"text/tabwriter"
)

// PrintJSON writes v as indented JSON followed by a newline.
func PrintJSON(w io.Writer, v any) error {
	b, err := json.MarshalIndent(v, "", "  ")
	if err != nil {
		return fmt.Errorf("marshal json: %w", err)
	}
	if _, err := w.Write(append(b, '\n')); err != nil {
		return err
	}
	return nil
}

// PrintTable renders a slice of struct rows or a single struct via tabwriter.
// Fields with a `header:"..."` tag become columns; untagged fields are skipped.
func PrintTable(w io.Writer, v any) error {
	rows, err := reflectToRows(v)
	if err != nil {
		return err
	}
	if len(rows) == 0 {
		return nil
	}

	tw := tabwriter.NewWriter(w, 0, 0, 2, ' ', 0)
	headers := rowHeaders(rows[0])
	if _, err := fmt.Fprintln(tw, joinTabs(headers)); err != nil {
		return err
	}
	for _, row := range rows {
		if _, err := fmt.Fprintln(tw, joinTabs(row.Values())); err != nil {
			return err
		}
	}
	return tw.Flush()
}

type tableRow struct {
	headers []string
	values  []string
}

func (r tableRow) Values() []string { return r.values }

func rowHeaders(r tableRow) []string { return r.headers }

func joinTabs(parts []string) string {
	out := ""
	for i, p := range parts {
		if i > 0 {
			out += "\t"
		}
		out += p
	}
	return out
}

// reflectToRows converts v (struct or slice of struct) into rows of strings.
func reflectToRows(v any) ([]tableRow, error) {
	rv := reflect.ValueOf(v)
	switch rv.Kind() {
	case reflect.Slice:
		out := make([]tableRow, 0, rv.Len())
		for i := 0; i < rv.Len(); i++ {
			row, err := structToRow(rv.Index(i))
			if err != nil {
				return nil, err
			}
			out = append(out, row)
		}
		return out, nil
	case reflect.Struct:
		row, err := structToRow(rv)
		if err != nil {
			return nil, err
		}
		return []tableRow{row}, nil
	case reflect.Ptr:
		if rv.IsNil() {
			return nil, nil
		}
		return reflectToRows(rv.Elem().Interface())
	default:
		return nil, fmt.Errorf("PrintTable: unsupported kind %s", rv.Kind())
	}
}

func structToRow(rv reflect.Value) (tableRow, error) {
	if rv.Kind() == reflect.Ptr {
		if rv.IsNil() {
			return tableRow{}, nil
		}
		rv = rv.Elem()
	}
	if rv.Kind() != reflect.Struct {
		return tableRow{}, fmt.Errorf("PrintTable: expected struct, got %s", rv.Kind())
	}
	rt := rv.Type()
	var headers, values []string
	for i := 0; i < rt.NumField(); i++ {
		f := rt.Field(i)
		header := f.Tag.Get("header")
		if header == "" {
			continue
		}
		headers = append(headers, header)
		values = append(values, fmt.Sprintf("%v", rv.Field(i).Interface()))
	}
	return tableRow{headers: headers, values: values}, nil
}
```

func TestAPIClient_RefreshOn401(t *testing.T) {
	var calls int
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		switch r.URL.Path {
		case "/auth/refresh":
			body := struct {
				User struct {
					ID      int64 `json:"id"`
					IsAdmin bool  `json:"is_admin"`
				} `json:"user"`
				Tokens struct {
					AccessToken  string `json:"access_token"`
					RefreshToken string `json:"refresh_token"`
				} `json:"tokens"`
			}{}
			body.Tokens.AccessToken = "newaccess"
			body.Tokens.RefreshToken = "newrefresh"
			body.User.ID = 1
			_ = json.NewEncoder(w).Encode(body)
		default:
			calls++
			authH := r.Header.Get("Authorization")
			if authH == "Bearer expired" && calls == 1 {
				http.Error(w, "expired", http.StatusUnauthorized)
				return
			}
			require.Equal(t, "Bearer newaccess", authH)
			_, _ = w.Write([]byte(`{"ok":true}`))
		}
	}))
	defer srv.Close()

	store := newMemoryStore(&session.Session{
		ServerURL: srv.URL, AccessToken: "expired", RefreshToken: "rtok",
	})
	c := apiclient.New(srv.URL, store)
	resp, err := c.DoJSON(context.Background(), http.MethodGet, "/protected", nil)
	require.NoError(t, err)
	defer resp.Body.Close()
	require.Equal(t, http.StatusOK, resp.StatusCode)

	saved := store.Get()
	require.Equal(t, "newaccess", saved.AccessToken)
	require.Equal(t, "newrefresh", saved.RefreshToken)
}

// memoryStore is a session.Store implementation backed by an in-process value.
type memoryStore struct{ s *session.Session }

func newMemoryStore(s *session.Session) *memoryStore { return &memoryStore{s: s} }

func (m *memoryStore) Load() (*session.Session, error) {
	if m.s == nil {
		return nil, session.ErrNoSession
	}
	cp := *m.s

// Package apiclient is the CLI's HTTP client to a Linktheca server. It signs
// requests with the access token from the session store and transparently
// refreshes on 401.
package apiclient

import (
	"bytes"
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net/http"
	"strings"
	"time"

	"github.com/ismd/linktheca/cmd/linktheca/internal/cli/session"
)

type Client struct {
	baseURL string
	http    *http.Client
	store   session.Store
}

func New(baseURL string, store session.Store) *Client {
	return &Client{
		baseURL: strings.TrimSuffix(baseURL, "/"),
		http:    &http.Client{Timeout: 30 * time.Second},
		store:   store,
	}
}

// DoJSON sends a JSON request with the access token attached. On 401 it
// refreshes once and retries. The caller is responsible for closing resp.Body.
func (c *Client) DoJSON(ctx context.Context, method, path string, body any) (*http.Response, error) {
	resp, err := c.send(ctx, method, path, body, true)
	if err != nil {
		return nil, err
	}
	if resp.StatusCode != http.StatusUnauthorized {
		return resp, nil
	}

	// Refresh attempt.
	resp.Body.Close()
	if err := c.refresh(ctx); err != nil {
		return nil, fmt.Errorf("session expired, run `linktheca auth login`: %w", err)
	}
	return c.send(ctx, method, path, body, true)
}

// PostJSON is a convenience: POST + JSON-decode the response body into out.
// Returns the raw response so callers can inspect status if they need to.
func (c *Client) PostJSON(ctx context.Context, path string, body, out any) error {
	resp, err := c.DoJSON(ctx, http.MethodPost, path, body)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		return decodeServerError(resp)
	}
	if out == nil {
		return nil
	}
	if err := json.NewDecoder(resp.Body).Decode(out); err != nil {
		return fmt.Errorf("decode response: %w", err)
	}
	return nil
}

// GetJSON: GET + decode into out.
func (c *Client) GetJSON(ctx context.Context, path string, out any) error {
	resp, err := c.DoJSON(ctx, http.MethodGet, path, nil)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		return decodeServerError(resp)
	}
	if out == nil {
		return nil
	}
	return json.NewDecoder(resp.Body).Decode(out)
}

func (c *Client) send(ctx context.Context, method, path string, body any, withAuth bool) (*http.Response, error) {
	var rdr io.Reader
	if body != nil {
		b, err := json.Marshal(body)
		if err != nil {
			return nil, fmt.Errorf("marshal body: %w", err)
		}
		rdr = bytes.NewReader(b)
	}

	req, err := http.NewRequestWithContext(ctx, method, c.baseURL+path, rdr)
	if err != nil {
		return nil, fmt.Errorf("build request: %w", err)
	}
	if body != nil {
		req.Header.Set("Content-Type", "application/json")
	}

	if withAuth {
		s, err := c.store.Load()
		if err != nil {
			return nil, fmt.Errorf("not logged in (run `linktheca auth login`): %w", err)
		}
		if s.AccessToken != "" {
			req.Header.Set("Authorization", "Bearer "+s.AccessToken)
		}
	}

	return c.http.Do(req)
}

func (c *Client) refresh(ctx context.Context) error {
	s, err := c.store.Load()
	if err != nil {
		return err
	}
	if s.RefreshToken == "" {
		return errors.New("no refresh token")
	}

	resp, err := c.send(ctx, http.MethodPost, "/auth/refresh",
		map[string]string{"refresh_token": s.RefreshToken}, false)
	if err != nil {
		return fmt.Errorf("refresh request: %w", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return decodeServerError(resp)
	}

	var out struct {
		User struct {
			ID      int64 `json:"id"`
			IsAdmin bool  `json:"is_admin"`
		} `json:"user"`
		Tokens struct {
			AccessToken  string `json:"access_token"`
			RefreshToken string `json:"refresh_token"`
		} `json:"tokens"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
		return fmt.Errorf("decode refresh response: %w", err)
	}

	s.AccessToken = out.Tokens.AccessToken
	s.RefreshToken = out.Tokens.RefreshToken
	if out.User.ID != 0 {
		s.UserID = out.User.ID
		s.IsAdmin = out.User.IsAdmin
	}
	return c.store.Save(s)
}

// ServerError is returned for non-2xx responses.
type ServerError struct {
	StatusCode int
	Code       string
	Message    string
}

func (e *ServerError) Error() string {
	if e.Code != "" {
		return fmt.Sprintf("server %d %s: %s", e.StatusCode, e.Code, e.Message)
	}
	return fmt.Sprintf("server %d: %s", e.StatusCode, e.Message)
}

func decodeServerError(resp *http.Response) error {
	body, _ := io.ReadAll(io.LimitReader(resp.Body, 4096))
	var parsed struct {
		Error struct {
			Code    string `json:"code"`
			Message string `json:"message"`
		} `json:"error"`
	}
	_ = json.Unmarshal(body, &parsed)
	return &ServerError{
		StatusCode: resp.StatusCode,
		Code:       parsed.Error.Code,
		Message:    cmpStr(parsed.Error.Message, string(body)),
	}
}

func cmpStr(a, b string) string {
	if a != "" {
		return a
	}
	return b
}

// Logout: POST /auth/logout (authenticated, plus refresh_token in body).
func (c *Client) Logout(ctx context.Context) error {
	s, err := c.store.Load()
	if err != nil {
		return nil // already not logged in
	}
	resp, err := c.DoJSON(ctx, http.MethodPost, "/auth/logout", auth.RefreshRequest{RefreshToken: s.RefreshToken})
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusNoContent && (resp.StatusCode < 200 || resp.StatusCode >= 300) {
		return decodeServerError(resp)
	}
	return c.store.Clear()
}

package cmd

import (
	"fmt"

	"github.com/ismd/linktheca/cmd/linktheca/internal/cli/apiclient"

	"github.com/spf13/cobra"
)

type feedRowView struct {
	ID                   int64  `header:"ID"       json:"id"`
	URL                  string `header:"URL"      json:"url"`
	Kind                 string `header:"Kind"     json:"kind"`
	FetchIntervalSeconds int    `header:"Interval" json:"fetch_interval_seconds"`
}

func newRadarFeedAdd(env *cli.Env) *cobra.Command {
	var url, kind string
	var interval int
	c := &cobra.Command{
		Use:   "add",
		Short: "Register a global RSS/Atom feed (admin only)",
		RunE: func(cmd *cobra.Command, _ []string) error {
			if url == "" {
				return fmt.Errorf("--url is required")
			}
			store := session.NewFileStore(env.ConfigDir)
			client := apiclient.New(env.ServerURL, store)

			req := radar.AddFeedRequest{URL: url}
			if cmd.Flags().Changed("kind") {
				req.Kind = &kind
			}
			if cmd.Flags().Changed("interval") {
				req.FetchIntervalSeconds = &interval
			}
			feed, err := client.AddFeed(cmd.Context(), req)
			if err != nil {
				return err
			}
			row := feedRowView{
				ID: feed.ID, URL: feed.URL, Kind: feed.Kind,
				FetchIntervalSeconds: feed.FetchIntervalSeconds,
			}
			return printOne(env, &row)
		},
	}
	c.Flags().StringVar(&url, "url", "", "feed URL")
	c.Flags().StringVar(&kind, "kind", "rss", "feed kind: rss | atom")
	c.Flags().IntVar(&interval, "interval", 3600, "fetch interval in seconds (300..86400)")
	return c
}

func TestIntegrationCLIFullFlow(t *testing.T) {
	if testing.Short() {
		t.Skip("skipping integration test")
	}

	binDir := t.TempDir()
	binPath := filepath.Join(binDir, "linktheca")
	out, err := exec.Command("go", "build", "-o", binPath, "./cmd/linktheca").CombinedOutput()
	require.NoError(t, err, "build cli: %s", string(out))

	pool := testdb.New(t)
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	require.NoError(t, err)
	addr := "http://" + listener.Addr().String()

	cfg := &config.Config{
		HTTPAddr:               listener.Addr().String(),
		JWTSecret:              "test-secret-at-least-32-bytes-long-for-hmac",
		JWTAccessTTL:           15 * time.Minute,
		JWTRefreshTTL:          24 * time.Hour,
		RegistrationEnabled:    true,
		RadarEnabled:           true,
		RadarSchedulerInterval: time.Hour, // effectively disable periodic
		RadarMaxWorkers:        1,
		EmbeddingDim:           1024,
	}
	logger := slog.New(slog.NewTextHandler(io.Discard, nil))

	// Build server WITHOUT starting River — Radar HTTP routes only need
	// embedder + (nil) river client.
	emb := &embeddings.FakeEmbedder{Dim: 1024}
	srv := server.New(server.Deps{
		Config: cfg, Logger: logger, DB: pool,
		Radar: &server.RadarDeps{Embedder: emb}, // no river client; pipeline not exercised
	})
	srv.Addr = listener.Addr().String()

	go func() { _ = srv.Serve(listener) }()
	defer func() {
		ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		_ = srv.Shutdown(ctx)
	}()

	// Wait for /healthz.
	require.Eventually(t, func() bool {
		resp, err := http.Get(addr + "/healthz")
		if err != nil {
			return false
		}
		defer resp.Body.Close()
		return resp.StatusCode == http.StatusOK
	}, 5*time.Second, 50*time.Millisecond, "server didn't come up")

	configDir := t.TempDir()
	envVars := append(os.Environ(),
		"LINKTHECA_URL="+addr,
		"LINKTHECA_CONFIG_DIR="+configDir,
	)

	runCLI := func(t *testing.T, args ...string) string {
		t.Helper()
		cmd := exec.Command(binPath, args...)
		cmd.Env = envVars
		out, err := cmd.CombinedOutput()
		require.NoError(t, err, "%s: %s", args, string(out))
		return string(out)

		"--password=correcthorse",
		"--display-name=Admin")

	require.NoError(t, err)
	require.Equal(t, os.FileMode(0o600), info.Mode().Perm())

	// whoami should print the user.
	out = runCLI(t, "auth", "whoami")
	require.Contains(t, string(out), "admin@example.com")

	// Add a feed (admin).
	out = runCLI(t, "radar", "feed", "add", "--url=https://news.example/rss")
	require.Contains(t, string(out), "https://news.example/rss")

	out = runCLI(t, "radar", "topic", "add",
		"--name=ML",
		"--description=machine learning research and large language models")
	require.Contains(t, string(out), "ML")

	var feedID int64
	require.NoError(t, pool.QueryRow(context.Background(),
		`SELECT id FROM radar_feeds WHERE url=$1`, "https://news.example/rss").Scan(&feedID))

	// Subscribe.
	out = runCLI(t, "radar", "subscribe", "--feed-id="+itoa(feedID))
	require.Contains(t, string(out), "User")

	// Verify session contents and DB rows.
	s, err := session.Load(configDir)
	require.NoError(t, err)
	require.NotEmpty(t, s.AccessToken)
	require.NotEmpty(t, s.RefreshToken)
	require.True(t, s.IsAdmin)

	var nSub, nTopic int
	require.NoError(t, pool.QueryRow(context.Background(),
		`SELECT count(*) FROM radar_feed_subscriptions WHERE user_id=$1 AND feed_id=$2`,
		s.UserID, feedID).Scan(&nSub))
	require.Equal(t, 1, nSub)

	require.NoError(t, pool.QueryRow(context.Background(),
		`SELECT count(*) FROM radar_topics WHERE user_id=$1 AND embedding IS NOT NULL`,
		s.UserID).Scan(&nTopic))
	require.Equal(t, 1, nTopic)

func formatInt(i int64) string {
	// Strconv would do but importing it for one call is fine; inline keeps the
	// import list shorter where the rest of the file doesn't use it.
	if i == 0 {
		return "0"
	}
	const digits = "0123456789"
	neg := i < 0
	if neg {
		i = -i
	}
	var buf [20]byte
	pos := len(buf)
	for i > 0 {
		pos--
		buf[pos] = digits[i%10]
		i /= 10
	}
	if neg {
		pos--
		buf[pos] = '-'
	}
	return string(buf[pos:])
